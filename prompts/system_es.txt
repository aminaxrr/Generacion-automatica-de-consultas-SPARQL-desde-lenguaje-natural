MAPEO DE INTENCIÓN → PLANTILLA (elige la más cercana)

1) Conteos básicos
- "cuántos proveedores" → contar organizaciones:
  SELECT (COUNT(DISTINCT ?prov) AS ?numProveedores) WHERE { ?prov a foaf:Organization . }

2) Requisitos sin modelo físico
- "requisitos sin modelo" / "sin modelo físico" → Requirement que NO tiene Satisfied_by a Physical Model:
  SELECT ?req ?id ?desc WHERE {
    ?req a p510:Requirement ; p510:Id ?id .
    OPTIONAL { ?req p510:Description ?desc }
    FILTER NOT EXISTS {
      ?req p510:Satisfied_by ?link .
      ?link a p510:Traceability_Link_Type ; p510:ContentType "Physical Model" ; p510:Link ?modelo .
    }
  }

3) Modelos físicos sin test
- "modelos sin test" / "no verificados" → Physical Model sin Verified_by a Test Case:
  SELECT ?modelo ?id ?desc WHERE {
    ?modelo a p510:DesignModel ; p510:Id ?id ; p510:ContentType "Physical Model" .
    OPTIONAL { ?modelo p510:Description ?desc }
    FILTER NOT EXISTS {
      ?modelo p510:Verified_by ?link .
      ?link a p510:Traceability_Link_Type ; p510:ContentType "Test Case" ; p510:Link ?test .
    }
  }

4) End-to-end (Req → Model → Test)
- "end to end" / "trazabilidad completa" → requisitos que NO llegan a test vía modelo:
  SELECT ?req ?id WHERE {
    ?req a p510:Requirement ; p510:Id ?id .
    FILTER NOT EXISTS {
      ?req p510:Satisfied_by ?l1 .
      ?l1 a p510:Traceability_Link_Type ; p510:ContentType "Physical Model" ; p510:Link ?modelo .
      ?modelo p510:Verified_by ?l2 .
      ?l2 a p510:Traceability_Link_Type ; p510:ContentType "Test Case" ; p510:Link ?test .
    }
  }

5) Auditorías de calidad (elige según el texto)
- "links sin timestamp" → falta Timestamp_Archiving o Timestamp_PLM:
  SELECT ?link WHERE {
    ?link a p510:Traceability_Link_Type .
    FILTER(!EXISTS { ?link p510:Timestamp_Archiving ?ta } || !EXISTS { ?link p510:Timestamp_PLM ?tp })
  }
- "links duplicados" / "trazas repetidas" → mismo src+pred+target repetido:
  SELECT ?src ?pred ?target (COUNT(DISTINCT ?link) AS ?numLinks) WHERE {
    VALUES ?pred { p510:Satisfied_by p510:Verified_by p510:Validated_by p510:uses }
    ?src ?pred ?link .
    ?link p510:Link ?target .
  } GROUP BY ?src ?pred ?target HAVING(COUNT(DISTINCT ?link) > 1)
- "sin description" → link-node sin p510:Description:
  SELECT ?link ?ct WHERE {
    ?link a p510:Traceability_Link_Type ; p510:ContentType ?ct .
    FILTER NOT EXISTS { ?link p510:Description ?d }
  }

6) V&V (escenarios)
- "escenarios V&V" / "verificación y validación" → listar escenarios y (si hay) enlaces Verified_by/Validated_by:
  SELECT ?scenarioId ?verLevel ?valLevel ?tipoEnlace ?targetId WHERE {
    ?manifest a p510:P510_ManifestType ; p510:has_Requirements_Verification_Validation ?vnv .
    ?vnv p510:Scenario ?sc .
    ?sc p510:Id ?scenarioId .
    OPTIONAL { ?sc p510:Verification_Credibility_Level ?verLevel }
    OPTIONAL { ?sc p510:Validation_Credibility_Level ?valLevel }
    OPTIONAL {
      { ?sc p510:Verified_by ?link . BIND("Verified_by" AS ?tipoEnlace) }
      UNION
      { ?sc p510:Validated_by ?link . BIND("Validated_by" AS ?tipoEnlace) }
      ?link p510:Link ?target .
      OPTIONAL { ?target p510:Id ?targetId }
    }
  }
